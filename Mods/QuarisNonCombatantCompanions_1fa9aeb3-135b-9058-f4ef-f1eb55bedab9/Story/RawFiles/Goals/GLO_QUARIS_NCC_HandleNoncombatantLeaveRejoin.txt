// Watch for noncombatants leaving and rejoining the party

// If someone leaves the party, then they might need to enter combat
// (E.g.: Laezel or Shadowheart in act 1; or Jaheira, Minsc, and maybe Wyll in act 3).
// So remove the non-combatant status whenever someone leaves the party.
// If/when the (former) non-combatant rejoins, then restore their non-combatant status.

// Note that summons get dismissed upon their owner leaving the party.
// Therefore, summons should be removed from the normal is-noncombatant DB upon leaving,
// but summons don't need any handling upon their hypothetical owner rejoining since that never happens.

// TODO: Add some sort of exception for the Lae'zel romance scene where she fights you?
// (Or does she technically leave the party? And so would that already be handled?)
// Any other exceptions for combat?

IF
CharacterLeftParty(_Character)
AND
QRY_QUARIS_NCC_IsNoncombatant(_Character)
THEN
// Leaving the party seems to make the character automatically move back on-stage
PROC_QUARIS_NCC_SetIsNoncombatantFalse(_Character);
PROC_QUARIS_NCC_IfNotSummonSetOutOfPartyNoncombatant(_Character);
RemoveStatus(_Character, "QUARIS_NCC_IS_OFFSTAGE_NONCOMBATANT_VISUAL_INDICATOR");

IF
CharacterJoinedParty(_Character)
AND
DB_QUARIS_NCC_OutOfPartyNoncombatants(_Character, _)
THEN
PROC_QUARIS_NCC_SetIsNoncombatantTrue(_Character);
NOT DB_QUARIS_NCC_OutOfPartyNoncombatants(_Character, 0);
NOT DB_QUARIS_NCC_OutOfPartyNoncombatants(_Character, 1);

IF
CharacterJoinedParty(_Character)
AND
DB_QUARIS_NCC_OutOfPartyNoncombatants(_Character, 0)
THEN
PROC_QUARIS_NCC_HandleKnockedDownBugAfterMovingBackOnstage(_Character);

// Set out-of-party noncombatant

PROC
PROC_QUARIS_NCC_IfNotSummonSetOutOfPartyNoncombatant((CHARACTER)_Character)
AND
IsSummon(_Character, 0)
THEN
PROC_QUARIS_NCC_IfWasOnstageSetOutOfPartyNoncombatantWasOnstage(_Character);
PROC_QUARIS_NCC_IfWasOffstageSetOutOfPartyNoncombatantWasOffstage(_Character);

PROC
PROC_QUARIS_NCC_IfWasOnstageSetOutOfPartyNoncombatantWasOnstage((CHARACTER)_Character)
AND
// This DB *might* be out-of-date by a frame or two?
// But it's ok if this check is not airtight; the knocked-down bug can still be
// remedied by putting the character off-stage and bringing them back again.
DB_QUARIS_NCC_AreNoncombatantsOffstage(0)
THEN
DB_QUARIS_NCC_OutOfPartyNoncombatants(_Character, 1);

PROC
PROC_QUARIS_NCC_IfWasOffstageSetOutOfPartyNoncombatantWasOffstage((CHARACTER)_Character)
AND
// This DB *might* be out-of-date by a frame or two?
// But it's ok if this check is not airtight; the knocked-down bug can still be
// remedied by putting the character off-stage and bringing them back again.
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
THEN
DB_QUARIS_NCC_OutOfPartyNoncombatants(_Character, 0);
ApplyStatus(_Character, "QUARIS_NCC_RECENTLY_MOVED_ON_STAGE_VISUAL_INDICATOR", 6.0);
EXITSECTION
