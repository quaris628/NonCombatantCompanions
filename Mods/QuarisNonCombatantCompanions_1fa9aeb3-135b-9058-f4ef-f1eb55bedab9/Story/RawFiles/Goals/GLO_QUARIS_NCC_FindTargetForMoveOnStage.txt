SubGoalCombiner SGC_AND
INITSECTION
DB_QUARIS_NCC_TargetForMovingNoncombatantOnStage((CHARACTER)NULL_00000000-0000-0000-0000-000000000000, (CHARACTER)NULL_00000000-0000-0000-0000-000000000000);
KBSECTION
// Determine where noncombatants should go after combat:
// If noncombatant grouped with the host avatar, go to host avatar
// If noncombatant grouped with one or more avatars and none are the host, pick any grouped avatar to go to
// If noncombatant grouped with only non-avatar combatants, pick any grouped combatant to go to
// If noncombatant not grouped with any combatants, stay where they were before combat (but still move a little bit to avoid harmful surfaces if needed)

// This is probably the most complicated part of this mod's scripts

// host avatar
QRY
QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage((CHARACTER)_Noncombatant)
AND
GetHostCharacter(_HostAvatar)
AND
InSamePartyGroup(_Noncombatant, _HostAvatar, 1)
THEN
DB_QUARIS_NCC_TargetForMovingNoncombatantOnStage(_Noncombatant, _HostAvatar);
DebugLog("QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage returning host avatar");

// non-host avatars
QRY
QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage((CHARACTER)_Noncombatant)
AND
GetHostCharacter(_HostCharacter)
AND
InSamePartyGroup(_Noncombatant, _HostCharacter, 0)
AND
DB_Avatars(_Avatars)
AND
InSamePartyGroup(_Noncombatant, _Avatars, 1)
THEN
// TODO: can I (ideally consistently) pick a single character from multiple?
DB_QUARIS_NCC_TargetForMovingNoncombatantOnStage(_Noncombatant, _Avatars);
DebugLog("QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage returning non-host avatars");

// non-avatar combatants
QRY
QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage((CHARACTER)_Noncombatant)
AND
DB_Avatars(_Avatars)
AND
InSamePartyGroup(_Noncombatant, _Avatars, 0)
AND
DB_PartyMembers(_PartyMembers)
AND
InSamePartyGroup(_Noncombatant, _PartyMembers, 1)
AND
HasAppliedStatus(_PartyMembers, "QUARIS_NCC_CompanionIsNoncombatant", 0)
THEN
// TODO: can I (ideally consistently) pick a single character from multiple?
DB_QUARIS_NCC_TargetForMovingNoncombatantOnStage(_Noncombatant, _PartyMembers);
DebugLog("QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage returning non-avatar combatants");

// myself
QRY
QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage((CHARACTER)_Noncombatant)
AND
DB_PartyMembers(_PartyMembers)
AND
InSamePartyGroup(_Noncombatant, _PartyMembers, 0)
AND
HasAppliedStatus(_PartyMembers, "QUARIS_NCC_CompanionIsNoncombatant", 1)
THEN
DB_QUARIS_NCC_TargetForMovingNoncombatantOnStage(_Noncombatant, _Noncombatant);
DebugLog("QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage returning myself");

// clear the DB used for return values
QRY
QRY_QUARIS_NCC_ClearTargetForMovingNoncombatantOnStage((CHARACTER)_Noncombatant)
AND
DB_PartyMembers(_PartyMembers)
AND
InSamePartyGroup(_Noncombatant, _PartyMembers, 1)
THEN
NOT DB_QUARIS_NCC_TargetForMovingNoncombatantOnStage(_Noncombatant, _PartyMembers);
EXITSECTION

ENDEXITSECTION
