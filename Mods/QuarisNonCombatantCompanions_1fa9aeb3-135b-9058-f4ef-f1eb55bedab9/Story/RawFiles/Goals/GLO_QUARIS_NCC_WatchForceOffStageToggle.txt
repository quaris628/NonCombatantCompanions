DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(0);

KBSECTION
// Watch for when the players manually toggle whether noncombatants are forced off-stage

// The toggle adds/removes hidden technical statuses that trigger this code,
// instead of simply adding/removing a single status directly.
// This is so that...
//  - This code can ensure the status won't be added if it's already there
//  - This code can differentiate between the toggle being changed by the user
//    vs. the visual indicator status being added/removed programmatically

// Watch hidden technical statuses => set flags

IF
StatusApplied(_Object, "QUARIS_NCC_NoncombatantsAreForcedOffstage_HiddenTechnicalAdd", _, _)
THEN
RemoveStatus(_Object, "QUARIS_NCC_NoncombatantsAreForcedOffstage_HiddenTechnicalAdd");
DB_QUARIS_NCC_CharacterHasForceNoncombatantsOffstageToggleOn((CHARACTER)_Object);
NOT DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(0);
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(1);

IF
StatusApplied(_Object, "QUARIS_NCC_NoncombatantsAreForcedOffstage_HiddenTechnicalRemove", _, _)
THEN
RemoveStatus(_Object, "QUARIS_NCC_NoncombatantsAreForcedOffstage_HiddenTechnicalRemove");
NOT DB_QUARIS_NCC_CharacterHasForceNoncombatantsOffstageToggleOn((CHARACTER)_Object);
NOT DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(1);
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(0);

// Watch flags => give/revoke visual status

IF
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(1)
AND
DB_PartyMembers(_PartyMember)
AND
IsSummon(_PartyMember, 0)
AND
IsPartyFollower(_PartyMember, 0)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_NoncombatantsAreForcedOffstage_VisualIndicator", 0)
THEN
ApplyStatus(_PartyMember, "QUARIS_NCC_NoncombatantsAreForcedOffstage_VisualIndicator", -1.0);

IF
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(0)
AND
DB_PartyMembers(_PartyMember)
THEN
RemoveStatus(_PartyMember, "QUARIS_NCC_NoncombatantsAreForcedOffstage_VisualIndicator");
EXITSECTION

ENDEXITSECTION
