INITSECTION

KBSECTION
// Handle syncing/displaying noncombatant status while off-stage

// Move off-stage => give status

IF
WentOnStage(_Object, 0)
AND
QRY_QUARIS_NCC_IsNoncombatant((CHARACTER)_Object)
THEN
ApplyStatus((CHARACTER)_Object, "QUARIS_NCC_IS_NONCOMBATANT_MAIN", -1.0);
// Piggyback the is-offstage-noncombatant status here,
// so that it's applied after the is-noncombatant status regardless of script names
ApplyStatus((CHARACTER)_Object, "QUARIS_NCC_IS_OFFSTAGE_NONCOMBATANT_VISUAL_INDICATOR", -1.0);

// Moving back on-stage doesn't need any special handling; the noncombatant status just persists on its own
EXITSECTION

ENDEXITSECTION
