SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
IF
DB_QUARIS_NCC_AreNoncombatantsOffstage(0)
AND
DB_PartyMembers(_Noncombatant)
AND
HasAppliedStatus(_Noncombatant, "QUARIS_NCC_IsNoncombatant_Main", 1)
AND
QRY_QUARIS_NCC_GetTargetForMovingNoncombatantOnStage(_Noncombatant)
AND
DB_QUARIS_NCC_TargetForMovingNoncombatantOnStage(_Noncombatant, _TargetCharacter)
AND
QRY_QUARIS_NCC_ClearTargetForMovingNoncombatantOnStage(_Noncombatant)
AND
HasAppliedStatus(_Noncombatant, "QUARIS_NCC_IsNoncombatant_Main", 1)
AND
GetPosition(_TargetCharacter, _TargetX, _TargetY, _TargetZ)
AND
FindValidPosition(_TargetX, _TargetY, _TargetZ, 10.0, _Noncombatant, 1, _ValidX, _ValidY, _ValidZ)
THEN
// The last few arguments for this TeleportToPosition call are:
//  - TeleportLinkedCharacters = 0
//  - TeleportPartyFollowers = 0
//  - TeleportSummons = 1
//  - LeaveCombat = 1
//  - SnapToGround = 1
TeleportToPosition(_Noncombatant, _ValidX, _ValidY, _ValidZ, "QUARIS_NCC_NoncombatantMovingOnstage", 0, 0, 1, 1, 1);
SetOnStage(_Noncombatant, 1);
// Workaround to this status not showing after the noncombatant comes back on stage
RemoveStatus(_Noncombatant, "QUARIS_NCC_IsNoncombatant_Main");
ApplyStatus(_Noncombatant, "QUARIS_NCC_IsNoncombatant_Main", -1.0);
PROC_QUARIS_NCC_PutOwnedSummonsOnstage(_Noncombatant);

PROC
PROC_QUARIS_NCC_PutOwnedSummonsOnstage((CHARACTER)_OwnerPartyMember)
AND
DB_Summons(_Summon)
AND
CharacterGetOwner((CHARACTER)_Summon, _OwnerPartyMember)
THEN
SetOnStage(_Summon, 1);
// Workaround to this status not showing after the noncombatant comes back on stage
RemoveStatus(_Summon, "QUARIS_NCC_IsNoncombatant_Main");
ApplyStatus(_Summon, "QUARIS_NCC_IsNoncombatant_Main", -1.0);

IF
CharacterLeftParty(_Character)
THEN
// Weirdly, someone leaving the party seems to make them move on-stage automatically?
// But do it explicitly too, just it case this helps handle something I'm not anticipating
SetOnStage(_Character, 1);
EXITSECTION

ENDEXITSECTION
