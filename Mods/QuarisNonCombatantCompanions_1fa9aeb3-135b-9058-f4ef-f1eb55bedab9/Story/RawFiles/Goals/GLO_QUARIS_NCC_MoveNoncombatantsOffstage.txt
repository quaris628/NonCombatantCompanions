Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// Move party members off-stage

IF
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
AND
DB_PartyMembers(_PartyMember)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_IS_NONCOMBATANT_MAIN", 1)
THEN
PROC_QUARIS_NCC_MoveOffstage(_PartyMember);

IF
StatusApplied(_Object, "QUARIS_NCC_IS_NONCOMBATANT_MAIN", _, _)
AND
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
AND
DB_PartyMembers((CHARACTER)_Object)
THEN
PROC_QUARIS_NCC_MoveOffstage((CHARACTER)_Object);

PROC
PROC_QUARIS_NCC_MoveOffstage((CHARACTER)_Character)
THEN
PROC_QUARIS_NCC_SetOffstageNoncombatantIsHiding(_Character);
RemoveStatus(_Character, "QUARIS_NCC_RECENTLY_MOVED_ON_STAGE_VISUAL_INDICATOR");
SetOnStage(_Character, 0);
EXITSECTION

ENDEXITSECTION
