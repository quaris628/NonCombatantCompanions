// Move party members and their summons off-stage

// Do NOT move "followers" off-stage, just in case it's important to the story that they be in combat
// (e.g. Florin in Wyrm's Rock, or Halsin in the Goblin camp)

IF
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
AND
DB_PartyMembers(_PartyMember)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_CompanionIsNoncombatant", 1)
THEN
PROC_QUARIS_NCC_MovePartyMemberAndSummonsOffStage(_PartyMember);

IF
StatusApplied(_Object, "QUARIS_NCC_CompanionIsNoncombatant", _, _)
AND
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
AND
DB_PartyMembers((CHARACTER)_Object)
THEN
PROC_QUARIS_NCC_MovePartyMemberAndSummonsOffStage((CHARACTER)_Object);

PROC
PROC_QUARIS_NCC_MovePartyMemberAndSummonsOffStage((CHARACTER)_PartyMember)
THEN
SetOnStage(_PartyMember, 0);
PROC_QUARIS_NCC_MoveSummonsOffStage(_PartyMember);

PROC
PROC_QUARIS_NCC_MoveSummonsOffStage((CHARACTER)_OwnerPartyMember)
AND
DB_Summons(_Summon)
AND
CharacterGetOwner((CHARACTER)_Summon, _OwnerPartyMember)
THEN
SetOnStage(_Summon, 0);
EXITSECTION

ENDEXITSECTION
