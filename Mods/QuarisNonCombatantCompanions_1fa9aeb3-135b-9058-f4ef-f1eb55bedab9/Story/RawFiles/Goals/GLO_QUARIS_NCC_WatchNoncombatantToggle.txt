// Watch for when the players manually toggle whether a party member is a noncombatant
// When they do, sync that party members' and their summons' noncombatant status

// The toggle adds/removes hidden technical statuses that trigger this code,
// instead of simply adding/removing a single status directly.
// This is so that...
//  - This code can ensure the status won't be added if it's already there
//  - This code can differentiate between the toggle being changed by the user
//    vs. the visual indicator status being added/removed programmatically
//    (such as the status being reapplied when the noncombatant moves off-stage)

// Handle triggers

IF
StatusApplied(_Object, "QUARIS_NCC_IS_NONCOMBATANT_HIDDEN_TECHNICAL_ADD", _, _)
THEN
RemoveStatus(_Object, "QUARIS_NCC_IS_NONCOMBATANT_HIDDEN_TECHNICAL_ADD");
PROC_QUARIS_NCC_HandleEnableNoncombatantAttempt((CHARACTER)_Object);

IF
StatusApplied(_Object, "QUARIS_NCC_IS_NONCOMBATANT_HIDDEN_TECHNICAL_REMOVE", _, _)
THEN
RemoveStatus(_Object, "QUARIS_NCC_IS_NONCOMBATANT_HIDDEN_TECHNICAL_REMOVE");
PROC_QUARIS_NCC_HandleDisableNoncombatantAttempt((CHARACTER)_Object);

// Noncombatant summons something => make summon noncombatant

IF
CharacterJoinedParty(_Summon)
AND
IsSummon(_Summon, 1)
AND
DB_PartyMembers(_OwnerPartyMember)
AND
CharacterGetOwner(_Summon, _OwnerPartyMember)
AND
QRY_QUARIS_NCC_IsNoncombatant(_OwnerPartyMember)
THEN
PROC_QUARIS_NCC_SetIsNoncombatantTrue(_Summon);

// Handle ui toggle enabled

PROC
PROC_QUARIS_NCC_HandleEnableNoncombatantAttempt((CHARACTER)_Companion)
AND
NOT QRY_QUARIS_NCC_IsNoncombatant(_Companion)
THEN
PROC_QUARIS_NCC_IfNoCombatMakeNoncombatant(_Companion);
PROC_QUARIS_NCC_IfCombatToggleNoncombatant(_Companion);

PROC
PROC_QUARIS_NCC_IfNoCombatMakeNoncombatant((CHARACTER)_Companion)
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(0)
AND
NOT QRY_QUARIS_NCC_IsNoncombatant(_Companion)
THEN
PROC_QUARIS_NCC_SetIsNoncombatantTrue(_Companion);
PROC_QUARIS_NCC_MakeSummonsNoncombatants(_Companion);

PROC
PROC_QUARIS_NCC_MakeSummonsNoncombatants((CHARACTER)_OwnerCompanion)
AND
DB_Summons(_Summon)
AND
CharacterGetOwner((CHARACTER)_Summon, _OwnerCompanion)
AND
NOT QRY_QUARIS_NCC_IsNoncombatant(_Summon)
THEN
PROC_QUARIS_NCC_SetIsNoncombatantTrue(_Summon);

// Handle ui toggle disabled

PROC
PROC_QUARIS_NCC_HandleDisableNoncombatantAttempt((CHARACTER)_Companion)
AND
QRY_QUARIS_NCC_IsNoncombatant(_Companion)
THEN
PROC_QUARIS_NCC_IfNoCombatMakeCombatant(_Companion);
PROC_QUARIS_NCC_IfCombatToggleNoncombatant(_Companion);

PROC
PROC_QUARIS_NCC_IfNoCombatMakeCombatant((CHARACTER)_Companion)
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(0)
AND
QRY_QUARIS_NCC_IsNoncombatant(_Companion)
THEN
PROC_QUARIS_NCC_SetIsNoncombatantFalse(_Companion);
PROC_QUARIS_NCC_MakeSummonsCombatants(_Companion);

PROC
PROC_QUARIS_NCC_MakeSummonsCombatants((CHARACTER)_OwnerCompanion)
AND
DB_Summons(_Summon)
AND
CharacterGetOwner((CHARACTER)_Summon, _OwnerCompanion)
AND
QRY_QUARIS_NCC_IsNoncombatant(_Summon)
THEN
PROC_QUARIS_NCC_SetIsNoncombatantFalse(_Summon);

// Toggle noncombatant passive
// (Intended for reverting back to an old state, so that the
// UI toggle matches the internally-tracked source of truth)

PROC
PROC_QUARIS_NCC_IfCombatToggleNoncombatant((CHARACTER)_Companion)
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(1)
THEN
TogglePassive(_Companion, "QUARIS_NCC_IsNoncombatantToggle");
// This ^ will trigger QUARIS_NCC_IS_NONCOMBATANT_HIDDEN_TECHNICAL_ADD
// or QUARIS_NCC_IS_NONCOMBATANT_HIDDEN_TECHNICAL_REMOVE being applied.
// That should do nothing, since PROC_QUARIS_NCC_HandleEnableNoncombatantAttempt
// and PROC_QUARIS_NCC_HandleDisableNoncombatantAttempt do nothing if the
// attempted change would already match the result of QRY_QUARIS_NCC_IsNoncombatant.
EXITSECTION
