// Watch for when the players manually toggle whether a party member is a noncombatant
// When they do, sync that party members' and their summons' noncombatant status

// The toggle adds/removes hidden technical statuses that trigger this code,
// instead of simply adding/removing a single status directly.
// This is so that...
//  - This code can ensure the status won't be added if it's already there
//  - This code can differentiate between the toggle being changed by the user
//    vs. the visual indicator status being added/removed programmatically

// If someone leaves the party, then they might need to enter combat
// (E.g.: Laezel or Shadowheart in act 1; or Jaheira, Minsc, and maybe Wyll in act 3).
// So remove the non-combatant status whenever someone leaves the party.

// When (re)joining the party, always default to not combatant,
// to avoid being locked out of controlling the character
// in case a combat is occurring halfway across the map.
// (Plus it's much simpler to implement.)

// TODO: Add some sort of exception for the Lae'zel romance scene where she fights you?
// (Or does she technically leave the party? And so would that already be handled?)
// Any other exceptions for combat?

// Watch hidden technical statuses => apply/remove main status

IF
StatusApplied(_Object, "QUARIS_NCC_IsNoncombatant_HiddenTechnicalAdd", _, _)
THEN
RemoveStatus(_Object, "QUARIS_NCC_IsNoncombatant_HiddenTechnicalAdd");
PROC_QUARIS_NCC_ApplyStatusIsNoncombatantToCompanion((CHARACTER)_Object);

IF
StatusApplied(_Object, "QUARIS_NCC_IsNoncombatant_HiddenTechnicalRemove", _, _)
THEN
RemoveStatus(_Object, "QUARIS_NCC_IsNoncombatant_HiddenTechnicalRemove");
PROC_QUARIS_NCC_RemoveStatusIsNoncombatantFromCompanion((CHARACTER)_Object);

// Noncombatant summons something => make summon a noncombatant too

IF
CharacterJoinedParty(_Summon)
AND
IsSummon(_Summon, 1)
AND
DB_PartyMembers(_OwnerPartyMember)
AND
CharacterGetOwner(_Summon, _OwnerPartyMember)
AND
HasAppliedStatus(_OwnerPartyMember, "QUARIS_NCC_IsNoncombatant_Main", 1)
THEN
ApplyStatus(_Summon, "QUARIS_NCC_IsNoncombatant_Main", -1.0);
ApplyStatus(_Summon, "QUARIS_NCC_IsNoncombatant_CombatLogVisual", -1.0);

// Character leaves party => ensure they're a combatant

IF
CharacterLeftParty(_Character)
THEN
PROC_QUARIS_NCC_SetNoncombatantToggleOff(_Character);

// Apply main status

PROC
PROC_QUARIS_NCC_ApplyStatusIsNoncombatantToCompanion((CHARACTER)_Companion)
THEN
PROC_QUARIS_NCC_IfNoCombatApplyStatusIsNoncombatantToCompanion(_Companion);
PROC_QUARIS_NCC_IfCombatUndoNoncombatantToggle(_Companion);

PROC
PROC_QUARIS_NCC_IfNoCombatApplyStatusIsNoncombatantToCompanion((CHARACTER)_Companion)
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(0)
AND
HasAppliedStatus(_Companion, "QUARIS_NCC_IsNoncombatant_Main", 0)
THEN
ApplyStatus(_Companion, "QUARIS_NCC_IsNoncombatant_Main", -1.0);
ApplyStatus(_Companion, "QUARIS_NCC_IsNoncombatant_CombatLogVisual", -1.0);
PROC_QUARIS_NCC_ApplyStatusIsNoncombatantToSummons(_Companion);

PROC
PROC_QUARIS_NCC_ApplyStatusIsNoncombatantToSummons((CHARACTER)_OwnerCompanion)
AND
DB_Summons(_Summon)
AND
CharacterGetOwner((CHARACTER)_Summon, _OwnerCompanion)
AND
HasAppliedStatus(_Summon, "QUARIS_NCC_IsNoncombatant_Main", 0)
THEN
ApplyStatus(_Summon, "QUARIS_NCC_IsNoncombatant_Main", -1.0);
ApplyStatus(_Summon, "QUARIS_NCC_IsNoncombatant_CombatLogVisual", -1.0);

PROC
PROC_QUARIS_NCC_IfCombatUndoNoncombatantToggle((CHARACTER)_Companion)
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(1)
THEN
TogglePassive(_Companion, "QUARIS_NCC_IsNoncombatantToggle");
// This ^ should also trigger QUARIS_NCC_IsNoncombatant_HiddenTechnicalRemove
// which shouldn't do anything else in this case

// Remove main status

PROC
PROC_QUARIS_NCC_RemoveStatusIsNoncombatantFromCompanion((CHARACTER)_Companion)
THEN
PROC_QUARIS_NCC_RemoveStatusIsNoncombatantFromSummons(_Companion);
RemoveStatus(_Companion, "QUARIS_NCC_IsNoncombatant_Main");
RemoveStatus(_Companion, "QUARIS_NCC_IsNoncombatant_CombatLogVisual");

PROC
PROC_QUARIS_NCC_SetNoncombatantToggleOff((CHARACTER)_Companion)
AND
HasAppliedStatus(_Companion, "QUARIS_NCC_IsNoncombatant_Main", 1)
THEN
TogglePassive(_Companion, "QUARIS_NCC_IsNoncombatantToggle");
// This ^ should also trigger QUARIS_NCC_IsNoncombatant_HiddenTechnicalRemove
// which should take care of removing QUARIS_NCC_IsNoncombatant_Main
// from the companion and their summons

PROC
PROC_QUARIS_NCC_RemoveStatusIsNoncombatantFromSummons((CHARACTER)_OwnerCompanion)
AND
DB_Summons(_Summon)
AND
CharacterGetOwner((CHARACTER)_Summon, _OwnerCompanion)
THEN
RemoveStatus(_Summon, "QUARIS_NCC_IsNoncombatant_Main");
RemoveStatus(_Summon, "QUARIS_NCC_IsNoncombatant_CombatLogVisual");
EXITSECTION

ENDEXITSECTION
