SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// Give the force-noncombatants-offstage toggle to all party members (but not summons or followers)

IF
LevelLoaded(_)
THEN
TimerLaunch("QUARIS_NCC_GiveForceOffstageToggle_InitDelay", 0);

IF
TimerFinished("QUARIS_NCC_GiveForceOffstageToggle_InitDelay")
AND
DB_PartyMembers(_PartyMembers)
THEN
PROC_QUARIS_NCC_GiveForceOffStageToggle(_PartyMembers);

// TODO: Test recruting companion
// TODO: Test multiplayer where an avatar joins partway through
// TODO: Test summoning a summon
// TODO: Test a follower joining the party
IF
CharacterJoinedParty(_Character)
THEN
DebugLog("CharacterJoinedParty is triggering PROC_QUARIS_NCC_GiveForceOffStageToggle");
PROC_QUARIS_NCC_GiveForceOffStageToggle(_Character);

PROC
PROC_QUARIS_NCC_GiveForceOffStageToggle((CHARACTER)_PartyMember)
AND
IsSummon(_PartyMember, 0)
AND
IsPartyFollower(_PartyMember, 0)
AND
HasPassive(_PartyMember, "QUARIS_NCC_ForceNoncombatantsOffstageToggle", 0)
THEN
AddPassive(_PartyMember, "QUARIS_NCC_ForceNoncombatantsOffstageToggle");
EXITSECTION

ENDEXITSECTION
