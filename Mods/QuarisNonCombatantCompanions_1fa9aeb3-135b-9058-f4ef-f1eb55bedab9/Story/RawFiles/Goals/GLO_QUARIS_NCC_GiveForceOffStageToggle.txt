Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// Give the force-noncombatants-offstage toggle to all party members (but not summons or followers)

// Triggers

IF
LevelLoaded(_)
THEN
TimerLaunch("QUARIS_NCC_GiveForceOffstageToggle_InitDelay", 0);

IF
TimerFinished("QUARIS_NCC_GiveForceOffstageToggle_InitDelay")
AND
DB_PartyMembers(_PartyMember)
THEN
PROC_QUARIS_NCC_InitForceOffstageToggle(_PartyMember);

IF
CharacterJoinedParty(_PartyMember)
AND
DB_PartyMembers(_PartyMember)
THEN
PROC_QUARIS_NCC_InitForceOffstageToggle(_PartyMember);

// Set up a party member

PROC
PROC_QUARIS_NCC_InitForceOffstageToggle((CHARACTER)_PartyMember)
AND
IsSummon(_PartyMember, 0)
AND
IsPartyFollower(_PartyMember, 0)
THEN
PROC_QUARIS_NCC_GiveForceOffstageToggle(_PartyMember);
// Sometimes the above procedure doesn't do any programmatic toggling
// and the visual indicator status still needs to be synced
PROC_QUARIS_NCC_SyncForcedOffstageVisualIndicator_ApplyIfOn(_PartyMember);
PROC_QUARIS_NCC_SyncForcedOffstageVisualIndicator_RemoveIfOff(_PartyMember);

// Give toggle

// no passive, force on, (no tracking) => give passive, toggle
// no passive, force off, (no tracking) => give it
// passive, force on, no tracking => toggle
// passive, force on, tracking => -
// passive, force off, no tracking => -
// passive, force off, tracking => toggle

PROC
PROC_QUARIS_NCC_GiveForceOffstageToggle((CHARACTER)_PartyMember)
THEN
PROC_QUARIS_NCC_GiveForceOffstageTogglePassiveIfNotPresent(_PartyMember);
PROC_QUARIS_NCC_ToggleForceNoncombatantsOffstageIfTrackedOffAndForceOn(_PartyMember);
PROC_QUARIS_NCC_ToggleForceNoncombatantsOffstageIfTrackedOnAndForceOff(_PartyMember);

PROC
PROC_QUARIS_NCC_GiveForceOffstageTogglePassiveIfNotPresent((CHARACTER)_PartyMember)
AND
HasPassive(_PartyMember, "QUARIS_NCC_ForceNoncombatantsOffstageToggle", 0)
THEN
AddPassive(_PartyMember, "QUARIS_NCC_ForceNoncombatantsOffstageToggle");
// I believe this would always be not set already, but sync it just in case
NOT DB_QUARIS_NCC_CharacterHasForceNoncombatantsOffstageToggleOn(_PartyMember);

PROC
PROC_QUARIS_NCC_ToggleForceNoncombatantsOffstageIfTrackedOffAndForceOn((CHARACTER)_PartyMember)
AND
NOT DB_QUARIS_NCC_CharacterHasForceNoncombatantsOffstageToggleOn(_PartyMember)
AND
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(1)
THEN
TogglePassive(_PartyMember, "QUARIS_NCC_ForceNoncombatantsOffstageToggle");
// Toggling this ^ should also apply QUARIS_NCC_NoncombatantsAreForcedOffstage_HiddenTechnicalAdd
// which should take care of syncing the visual indicator status, etc.

PROC
PROC_QUARIS_NCC_ToggleForceNoncombatantsOffstageIfTrackedOnAndForceOff((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_CharacterHasForceNoncombatantsOffstageToggleOn(_PartyMember)
AND
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(0)
THEN
TogglePassive(_PartyMember, "QUARIS_NCC_ForceNoncombatantsOffstageToggle");
// Toggling this ^ should also apply QUARIS_NCC_NoncombatantsAreForcedOffstage_HiddenTechnicalRemove
// which should take care of syncing the visual indicator status, etc.

// Remove toggle

PROC
PROC_QUARIS_NCC_RemoveForceOffstageToggle((CHARACTER)_PartyMember)
THEN
RemovePassive(_PartyMember, "QUARIS_NCC_ForceNoncombatantsOffstageToggle");
RemoveStatus(_PartyMember, "QUARIS_NCC_NoncombatantsAreForcedOffstage_VisualIndicator");

// Sync visual indicator

PROC
PROC_QUARIS_NCC_SyncForcedOffstageVisualIndicator_ApplyIfOn((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(1)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_NoncombatantsAreForcedOffstage_VisualIndicator", 0)
THEN
ApplyStatus(_PartyMember, "QUARIS_NCC_NoncombatantsAreForcedOffstage_VisualIndicator", -1.0);

PROC
PROC_QUARIS_NCC_SyncForcedOffstageVisualIndicator_RemoveIfOff((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_AreNoncombatantsForcedOffstage(0)
THEN
RemoveStatus(_PartyMember, "QUARIS_NCC_NoncombatantsAreForcedOffstage_VisualIndicator");
EXITSECTION

ENDEXITSECTION
