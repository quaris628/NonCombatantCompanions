Version 1
SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// Give the is-noncombatant toggle to all non-avatar party members

// TODO: Grant the is-noncombatant status to all summons of noncombatants too

IF
LevelLoaded(_)
THEN
TimerLaunch("QUARIS_NCC_CompanionIsNoncombatantToggle_InitDelay", 0);

IF
TimerFinished("QUARIS_NCC_CompanionIsNoncombatantToggle_InitDelay")
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(0)
THEN
PROC_QUARIS_NCC_GiveNoncombatantToggle();

IF
CharacterJoinedParty(_Character)
AND
DB_PartyMembers(_Character)
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(0)
THEN
PROC_QUARIS_NCC_GiveNoncombatantToggle();

// Revoke the toggle while any combat is in progress

IF
DB_QUARIS_NCC_IsAnyCombatInProgress(1)
THEN
PROC_QUARIS_NCC_RevokeNoncombatantToggle();

IF
DB_QUARIS_NCC_IsAnyCombatInProgress(0)
THEN
PROC_QUARIS_NCC_GiveNoncombatantToggle();

// Procedures to give/revoke the toggle

PROC
PROC_QUARIS_NCC_GiveNoncombatantToggle()
AND
DB_PartyMembers(_PartyMembers)
AND
NOT DB_Avatars(_PartyMembers)
AND
IsSummon(_PartyMembers, 0)
AND
IsPartyFollower(_PartyMembers, 0)
AND
HasPassive(_PartyMembers, "QUARIS_NCC_CompanionIsNoncombatantToggle", 0)
THEN
AddPassive(_PartyMembers, "QUARIS_NCC_CompanionIsNoncombatantToggle");

PROC
PROC_QUARIS_NCC_RevokeNoncombatantToggle()
AND
DB_PartyMembers(_PartyMembers)
AND
HasPassive(_PartyMembers, "QUARIS_NCC_CompanionIsNoncombatantToggle", 1)
THEN
RemovePassive(_PartyMembers, "QUARIS_NCC_CompanionIsNoncombatantToggle");
EXITSECTION

ENDEXITSECTION
