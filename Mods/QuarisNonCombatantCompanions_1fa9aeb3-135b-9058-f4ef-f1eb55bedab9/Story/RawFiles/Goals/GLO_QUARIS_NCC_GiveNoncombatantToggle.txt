// Give the is-noncombatant toggle to all non-avatar party members (but not summons or followers)

// When (re)joining the party, always default to not combatant,
// to avoid being locked out of controlling the character
// in case a combat is occurring halfway across the map.
// (Plus it's much simpler to implement.)

// Game load => give toggle

IF
LevelLoaded(_)
THEN
TimerLaunch("QUARIS_NCC_IsNoncombatantToggle_InitDelay", 0);

IF
TimerFinished("QUARIS_NCC_IsNoncombatantToggle_InitDelay")
AND
DB_PartyMembers(_PartyMember)
THEN
PROC_QUARIS_NCC_GiveIsNoncombatantToggle(_PartyMember);

// Character joins party => give toggle

IF
CharacterJoinedParty(_PartyMember)
AND
DB_PartyMembers(_PartyMember)
THEN
PROC_QUARIS_NCC_GiveIsNoncombatantToggle(_PartyMember);


// Character leaves party => remove toggle
IF
CharacterLeftParty(_PartyMember)
THEN
PROC_QUARIS_NCC_RemoveIsNoncombatantToggle(_PartyMember);

// Give toggle (and set to the correct default)

PROC
PROC_QUARIS_NCC_GiveIsNoncombatantToggle((CHARACTER)_PartyMember)
AND
IsSummon(_PartyMember, 0)
AND
IsPartyFollower(_PartyMember, 0)
AND
NOT QRY_QUARIS_NCC_IsAvatar(_PartyMember)
AND
HasPassive(_PartyMember, "QUARIS_NCC_IsNoncombatantToggle", 0)
THEN
AddPassive(_PartyMember, "QUARIS_NCC_IsNoncombatantToggle");

// Remove toggle

PROC
PROC_QUARIS_NCC_RemoveIsNoncombatantToggle((CHARACTER)_PartyMember)
THEN
RemovePassive(_PartyMember, "QUARIS_NCC_IsNoncombatantToggle");
