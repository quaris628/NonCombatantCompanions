// Give the is-noncombatant toggle to all non-avatar party members (but not summons or followers)

// When (re)joining the party, always default to not combatant,
// to avoid being locked out of controlling the character
// in case a combat is occurring halfway across the map.
// (Plus it's much simpler to implement.)

// Game load => set up toggle

IF
LevelLoaded(_)
THEN
TimerLaunch("QUARIS_NCC_IsNoncombatantToggle_InitDelay", 0);

IF
TimerFinished("QUARIS_NCC_IsNoncombatantToggle_InitDelay")
AND
DB_PartyMembers(_PartyMember)
THEN
PROC_QUARIS_NCC_SetUpIsNoncombatantToggle(_PartyMember);

// Character joins party => set up toggle

IF
CharacterJoinedParty(_PartyMember)
AND
// probably unnecessary, but just in case of shennanigans with definitions not exactly matching...
DB_PartyMembers(_PartyMember)
THEN
PROC_QUARIS_NCC_SetUpIsNoncombatantToggle(_PartyMember);

// Set up toggle

PROC
PROC_QUARIS_NCC_SetUpIsNoncombatantToggle((CHARACTER)_PartyMember)
AND
IsSummon(_PartyMember, 0)
AND
IsPartyFollower(_PartyMember, 0)
AND
NOT QRY_QUARIS_NCC_IsAvatar(_PartyMember)
THEN
PROC_QUARIS_NCC_GiveIsNoncombatantToggleIfNotPresent(_PartyMember);
PROC_QUARIS_NCC_DisableIsNoncombatantToggleIfCombatInProgress(_PartyMember);

// Give toggle

PROC
PROC_QUARIS_NCC_GiveIsNoncombatantToggleIfNotPresent((CHARACTER)_PartyMember)
AND
HasPassive(_PartyMember, "QUARIS_NCC_IsNoncombatantToggle", 0)
THEN
AddPassive(_PartyMember, "QUARIS_NCC_IsNoncombatantToggle");

// Disable toggle if combat in progress

PROC
PROC_QUARIS_NCC_DisableIsNoncombatantToggleIfCombatInProgress((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_IsAnyCombatInProgress(1)
THEN
ApplyStatus(_PartyMember, "QUARIS_NCC_DISABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL", -1.0);
EXITSECTION

ENDEXITSECTION
