Version 1
SubGoalCombiner SGC_AND
INITSECTION
DB_QUARIS_NCC_IsAnyCombatInProgress(0);
KBSECTION
// Watch for whether any combat involving players is in progress

IF
CombatStarted(_Combat)
AND
QRY_QUARIS_NCC_IsAnyPartyMemberInCombat()
THEN
NOT DB_QUARIS_NCC_IsAnyCombatInProgress(0);
DB_QUARIS_NCC_IsAnyCombatInProgress(1);
PROC_QUARIS_NCC_DisableNoncombatantToggle();

IF
CombatEnded(_Combat)
AND
NOT QRY_QUARIS_NCC_IsAnyPartyMemberInCombat()
THEN
NOT DB_QUARIS_NCC_IsAnyCombatInProgress(1);
DB_QUARIS_NCC_IsAnyCombatInProgress(0);
PROC_QUARIS_NCC_EnableNoncombatantToggle();

QRY
QRY_QUARIS_NCC_IsAnyPartyMemberInCombat()
AND
DB_PartyMembers(_PartyMember)
AND
// Noncombatants might technically be in combat for a very short amount of time,
// before they are taken off stage in reaction to the combat having started?
// But we don't want to count that
HasAppliedStatus(_PartyMember, "QUARIS_NCC_IS_NONCOMBATANT_MAIN", 0)
AND
// I suspect IsInCombat would return 1 for narrative combats, but I couldn't
// find any documentation to confirm, and it's difficult to test.
// If it doesn't, then it might be necessary to check IsInNarrativeCombat here.
IsInCombat(_PartyMember, 1)
THEN
DB_NOOP(1);

PROC
PROC_QUARIS_NCC_DisableNoncombatantToggle()
AND
DB_PartyMembers(_PartyMember)
AND
HasPassive(_PartyMember, "QUARIS_NCC_IsNoncombatantToggle", 1)
THEN
ApplyStatus(_PartyMember, "QUARIS_NCC_DISABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL", -1.0);

PROC
PROC_QUARIS_NCC_EnableNoncombatantToggle()
AND
DB_PartyMembers(_PartyMember)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_DISABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL", 1)
THEN
RemoveStatus(_PartyMember, "QUARIS_NCC_DISABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL");
// Having OnStatusRemove in EnabledContext doesn't trigger the toggle re-enabling for some reason,
// but applying an empty status (to trigger OnStatusApplied) does trigger the toggle re-enabling
// so keep doing this as a workaround
ApplyStatus(_PartyMember, "QUARIS_NCC_ENABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL", 0.1);
RealtimeObjectTimerLaunch(_PartyMember, "QUARIS_NCC_ENABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL_RemoveDelay", 0);

IF
ObjectTimerFinished(_PartyMember, "QUARIS_NCC_ENABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL_RemoveDelay")
AND
IsOnStage(_PartyMember, 1)
THEN
RemoveStatus(_PartyMember, "QUARIS_NCC_ENABLE_NONCOMBATANT_TOGGLE_HIDDEN_TECHNICAL");
EXITSECTION

ENDEXITSECTION
