// Watch for whether any combat involving players is in progress

IF
CombatStarted(_Combat)
THEN
NOT DB_QUARIS_NCC_IsAnyCombatInProgress(0);
DB_QUARIS_NCC_IsAnyCombatInProgress(1);

IF
CombatEnded(_Combat)
AND
// Includes summons, followers, etc
DB_Players(_Players)
AND
NOT IsInCombat((GUIDSTRING)_Players, 1)
//AND
// Noncombatants might technically be in combat for a very short amount of time,
// before they are taken off stage in reaction to the combat having started.
// But we don't want to count that as "in combat" for these purposes.
//HasAppliedStatus(_Players, "QUARIS_NCC_CompanionIsNoncombatant", 0)
// TODO: What about IsInNarrativeCombat? What is the difference?
THEN
DebugLog("GLO_QUARIS_NCC_WatchCombat debug setting DB_QUARIS_NCC_IsAnyCombatInProgress(0)");
NOT DB_QUARIS_NCC_IsAnyCombatInProgress(1);
DB_QUARIS_NCC_IsAnyCombatInProgress(0);

THEN

CombatEnded(_Combat)
AND
// Includes summons, followers, etc
DB_Players(_Players)
