Version 1
SubGoalCombiner SGC_AND
INITSECTION
DB_QUARIS_NCC_IsAnyCombatInProgress(0);
KBSECTION
// Watch for whether any combat involving players is in progress

IF
CombatStarted(_Combat)
AND
QRY_QUARIS_NCC_IsAnyPartyMemberInCombat()
THEN
NOT DB_QUARIS_NCC_IsAnyCombatInProgress(0);
DB_QUARIS_NCC_IsAnyCombatInProgress(1);

IF
CombatEnded(_Combat)
AND
NOT QRY_QUARIS_NCC_IsAnyPartyMemberInCombat()
THEN
NOT DB_QUARIS_NCC_IsAnyCombatInProgress(1);
DB_QUARIS_NCC_IsAnyCombatInProgress(0);

QRY
QRY_QUARIS_NCC_IsAnyPartyMemberInCombat()
AND
DB_PartyMembers(_PartyMember)
AND
// Noncombatants might technically be in combat for a very short amount of time,
// before they are taken off stage in reaction to the combat having started?
// But we don't want to count that
HasAppliedStatus(_PartyMember, "QUARIS_NCC_IsNoncombatant_Main", 0)
AND
// I suspect IsInCombat would return 1 for narrative combats, but I couldn't
// find any documentation to confirm, and it's difficult to test.
// If it doesn't, then it might be necessary to check IsInNarrativeCombat here.
IsInCombat(_PartyMember, 1)
THEN
DB_NOOP(1);
EXITSECTION

ENDEXITSECTION
