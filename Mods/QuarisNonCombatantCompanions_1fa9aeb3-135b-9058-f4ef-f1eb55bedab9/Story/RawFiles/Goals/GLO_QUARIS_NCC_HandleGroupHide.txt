// Handle group hide

// Use separate DB to track whether someone is hiding, b/c HasAppliedStatus isn't reliable when off-stage

PROC
PROC_QUARIS_NCC_SetOffstagePartyMemberIsHiding((CHARACTER)_PartyMember)
AND
HasAppliedStatus(_PartyMember, "SNEAKING", 1)
THEN
DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember);

// Noncombatant moves offstage => make their hiding status persist

PROC
PROC_QUARIS_NCC_SyncHideStatusWhenNoncombatantsMoveOffstage()
AND
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
AND
DB_PartyMembers(_PartyMember)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_IsNoncombatant_Main", 1)
AND
DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember)
THEN
// I have no idea why, but BOTH of these ApplyStatus calls (one now and one delayed)
// are NECESSARY in order to successfully apply the status to an off-stage party member.
// If only one of these calls happens, then the status will not be applied.
// *shrugs vigorously*
ApplyStatus(_PartyMember, "SNEAKING", -1.0);
// TODO: Hande race conditions here better?
TimerLaunch("QUARIS_NCC_HideOffstageNoncombatants_DelayForSecondApplyStatus", 0);

IF
TimerFinished("QUARIS_NCC_HideOffstageNoncombatants_DelayForSecondApplyStatus")
AND
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
AND
DB_PartyMembers(_PartyMember)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_IsNoncombatant_Main", 1)
AND
DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember)
THEN
ApplyStatus(_PartyMember, "SNEAKING", -1.0);

// When group hide is used and fails for an offstage noncombatant => toggle their hiding state

IF
CastSpellFailed(_Object, "Shout_Hide", "shout", "", 0)
AND
HasAppliedStatus(_Object, "QUARIS_NCC_IsNoncombatant_Main", 1)
AND
DB_QUARIS_NCC_AreNoncombatantsOffstage(1)
THEN
PROC_QUARIS_NCC_IfOffstageNoncombatantHidingRemoveStatus((CHARACTER)_Object);
PROC_QUARIS_NCC_IfOffstageNoncombatantNotHidingApplyStatus((CHARACTER)_Object);
PROC_QUARIS_NCC_IfOffstageNoncombatantWasHidingSetIsHidingFalse((CHARACTER)_Object);
PROC_QUARIS_NCC_IfOffstageNoncombatantWasNotHidingSetIsHidingTrue((CHARACTER)_Object);

PROC
PROC_QUARIS_NCC_IfOffstageNoncombatantHidingRemoveStatus((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember)
THEN
// TODO: Does this vvv work? Or do I need to RemoveStatus a second time after a delay?
RemoveStatus(_PartyMember, "SNEAKING");
DB_QUARIS_NCC_OffstageNoncombatantWasHidingTemp(_PartyMember);
// TODO: handle race conditions here better
TimerLaunch("QUARIS_NCC_IfOffstageNoncombatantHidingRemoveStatus_DelayForSecondRemoveStatus", 0);

IF
TimerFinished("QUARIS_NCC_IfOffstageNoncombatantHidingRemoveStatus_DelayForSecondRemoveStatus")
AND
DB_QUARIS_NCC_OffstageNoncombatantWasHidingTemp(_PartyMember)
THEN
RemoveStatus(_PartyMember, "SNEAKING");
NOT DB_QUARIS_NCC_OffstageNoncombatantWasHidingTemp(_PartyMember);

PROC
PROC_QUARIS_NCC_IfOffstageNoncombatantNotHidingApplyStatus((CHARACTER)_PartyMember)
AND
NOT DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember)
THEN
// TODO: Does this vvv work? Or do I need to ApplyStatus a second time after a delay?
ApplyStatus(_PartyMember, "SNEAKING", -1.0);
DB_QUARIS_NCC_OffstageNoncombatantWasNotHidingTemp(_PartyMember);
// TODO: handle race conditions here better
TimerLaunch("QUARIS_NCC_IfOffstageNoncombatantNotHidingApplyStatus_DelayForSecondApplyStatus", 0);

IF
TimerFinished("QUARIS_NCC_IfOffstageNoncombatantNotHidingApplyStatus_DelayForSecondApplyStatus")
AND
DB_QUARIS_NCC_OffstageNoncombatantWasNotHidingTemp(_PartyMember)
THEN
PROC_ApplySneakingStatusIfNotAlreadyApplied(_PartyMember);
NOT DB_QUARIS_NCC_OffstageNoncombatantWasNotHidingTemp(_PartyMember);

PROC
PROC_ApplySneakingStatusIfNotAlreadyApplied((CHARACTER)_PartyMember)
AND
HasAppliedStatus(_PartyMember, "SNEAKING", 1)
THEN
ApplyStatus(_PartyMember, "SNEAKING", -1.0);

PROC
PROC_QUARIS_NCC_IfOffstageNoncombatantWasHidingSetIsHidingFalse((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_OffstageNoncombatantWasHidingTemp(_PartyMember)
THEN
NOT DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember);

PROC
PROC_QUARIS_NCC_IfOffstageNoncombatantWasNotHidingSetIsHidingTrue((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_OffstageNoncombatantWasNotHidingTemp(_PartyMember)
THEN
DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember);

// Noncombatant moves onstage => make their hiding status persist

IF
WentOnStage(_Object, 1)
AND
HasAppliedStatus(_Object, "QUARIS_NCC_IsNoncombatant_Main", 1)
THEN
PROC_QUARIS_NCC_IfWasHidingApplyStatus((CHARACTER)_Object);

PROC
PROC_QUARIS_NCC_SyncHideStatusWhenNoncombatantsMoveOnstage((CHARACTER)_PartyMember)
AND
HasAppliedStatus(_PartyMember, "QUARIS_NCC_IsNoncombatant_Main", 1)
THEN
PROC_QUARIS_NCC_IfWasHidingApplyStatus(_PartyMember);
PROC_QUARIS_NCC_IfWasNotHidingRemoveStatus(_PartyMember);
NOT DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember);

PROC
PROC_QUARIS_NCC_IfWasHidingApplyStatus((CHARACTER)_PartyMember)
AND
DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember)
THEN
ApplyStatus(_PartyMember, "SNEAKING", -1.0);
NOT DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember);

IF
TimerFinished("QUARIS_NCC_IfWasHidingApplyStatus_DelayForSecondApplyStatus")
AND
DB_QUARIS_NCC_NoncombatantWasHidingTemp(_PartyMember)
THEN
PROC_ApplySneakingStatusIfNotAlreadyApplied(_PartyMember);
NOT DB_QUARIS_NCC_NoncombatantWasHidingTemp(_PartyMember);

PROC
PROC_QUARIS_NCC_IfWasNotHidingRemoveStatus((CHARACTER)_PartyMember)
AND
NOT DB_QUARIS_NCC_OffstageNoncombatantIsHiding(_PartyMember)
THEN
RemoveStatus(_PartyMember, "SNEAKING");
